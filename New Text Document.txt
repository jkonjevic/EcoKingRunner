@echo off
setlocal
cd /d "%~dp0"

:: Create virtual environment if it doesn't exist
if not exist ".venv\Scripts\python.exe" (
    echo Creating virtual environment...
    py -3 -m venv .venv
)

:: Activate virtual environment
call ".venv\Scripts\activate.bat"

:: Set Playwright path and install dependencies
set PLAYWRIGHT_BROWSERS_PATH=0
python -m pip install --upgrade pip
python -m pip install -r requirements.txt pyinstaller
python -m playwright install chromium

:: Package application with PyInstaller
pyinstaller --noconfirm --clean --onedir --windowed --name EcoKingRunner --hidden-import ecoking_daily --collect-all playwright ecoking_launcher.py

pause
